1. EXECUTIVE SUMMARY & PROJECT VISION
This blueprint defines a production-grade, research-validated hybrid AI system for early-stage lung
cancer detection that simultaneously answers two clinical questions: (A) Is the current nodule
malignant? and (B) What is the patient's 5-year lung cancer risk? The system fuses two
independently proven deep learning models — MIT's Sybil (global volumetric risk scoring) and a
fine-tuned ResNet-50 (local nodule classification) — via a late-fusion Feature-Level MLP. This is a
Cascaded Multi-Scale Spatiotemporal Architecture.

2. PROBLEM STATEMENT & CLINICAL CONTEXT
Lung cancer is the leading cause of cancer mortality globally, responsible for ~1.8 million
deaths/year (WHO 2023). The five-year survival rate exceeds 55% if detected at Stage I, but drops
below 5% at Stage IV. The core diagnostic challenge is two-fold:
Problem A — Invisible Risk
Problem B — Nodule Ambiguity
Sybil showed that subtle volumetric CT patterns
predict future cancer before a detectable nodule
forms. Standard radiologist review misses these
diffuse vascular/texture signals. A 3D CNN operating
on the full LDCT volume is required.
When a nodule IS visible, classifying it as benign vs.
malignant is ambiguous even for experts.
Inter-radiologist agreement (kappa) is often <0.6. A
fine-tuned convolutional classifier on the cropped
ROI provides objective, reproducible classification.

4. GLOBAL RISK BRANCH — SYBIL (3D CNN)
Sybil was published by Mikhail Yala et al. (MIT CSAIL) in the Journal of Clinical Oncology (2023). It
processes an entire raw LDCT scan — without any human annotation — and outputs per-year cancer
risk probabilities for years 1 through 6. Its internal 3D convolutional encoder learns volumetric
patterns that precede macroscopic tumor formation.

import torch
from sybil import Sybil, Serie
class SybilFeatureExtractor(torch.nn.Module):
    def __init__(self, sybil_model):
        super().__init__()
        self.encoder  = sybil_model.encoder   # 3D CNN backbone
        self.pool     = sybil_model.pool       # Global avg-pool → 512-d
        # NOTE: Do NOT include risk_head — we want the embedding
    
    def forward(self, volume_tensor):
        # volume_tensor: (B, 1, D, H, W) — normalized LDCT
        feats = self.encoder(volume_tensor)    # (B, 512, d, h, w)
        embed = self.pool(feats)               # (B, 512, 1, 1, 1)
        return embed.flatten(1)                # (B, 512)

5. LOCAL MALIGNANCY BRANCH — ResNet-50
The ResNet-50 branch performs fine-grained, nodule-level classification. It receives a tightly
cropped 3D patch (64×64×64 voxels) around each candidate nodule detected by a nodule detector
(e.g., LungMask, nnU-Net, or LIDC bounding boxes). The backbone is pre-trained on ImageNet and
fine-tuned end-to-end on LIDC-IDRI annotations.
Backbone
ResNet-50 (torchvision.models.resnet50, pretrained=ImageNet)
Input
64×64×64 voxel float32 tensor, shape (B, 1, 64, 64, 64)
Modification
Replace conv1 kernel: (7×7×2D) → (7×7×7 3D). Replace all BatchNorm2d
→ BatchNorm3d
Fine-tune Data
LIDC-IDRI: 1,018 CT scans, 244,527 annotated nodules
Label Mapping
Mean radiologist score ≥ 3 → Malignant; < 3 → Benign
FC Head
AvgPool → Flatten → FC(2048, 256) → ReLU → Dropout(0.5) → FC(256,
2)
Feature Output
256-dim embedding (pre-final-softmax)
Augmentation
Random flip (x,y,z), rotation ±15°, elastic deform, intensity jitter

import torch
import torch.nn as nn
from torchvision.models import resnet50
class ResNet3D_Nodule(nn.Module):
    def __init__(self, embed_dim=256, num_classes=2):
        super().__init__()
        base  = resnet50(pretrained=True)
        
        # Adapt first conv layer: 2D → 3D
        self.conv1 = nn.Conv3d(1, 64, kernel_size=7, stride=2,
                               padding=3, bias=False)

        with torch.no_grad():

6. FEATURE FUSION LAYER & FINAL CLASSIFIER
Fusion vector dimension: 512 (Sybil) + 256 (ResNet) = 768-dimensional concatenated
embedding. This vector encodes both global volumetric risk context and local nodule
morphology simultaneously.
Complete HybridLungNet Class:
import torch
import torch.nn as nn
import torch.nn.functional as F
class HybridLungNet(nn.Module):
    """
    Spatiotemporal Hybrid Lung Cancer Detection Network
    Inputs : (volume_tensor, nodule_patch)
    Outputs: (malignancy_logits, risk_scores)
    """
    def __init__(self, sybil_extractor, resnet_nodule,
                 fusion_dim=384, num_classes=2, risk_years=6):
        super().__init__()
        self.global_branch = sybil_extractor   # output: 512-d (frozen)
        self.local_branch  = resnet_nodule     # output: 256-d
        
        in_dim = 512 + 256  # 768
        
        # Shared Fusion MLP
        self.fusion_mlp = nn.Sequential(
            nn.Linear(in_dim, fusion_dim),
            nn.LayerNorm(fusion_dim),
            nn.ReLU(inplace=True),
            nn.Dropout(p=0.5),
            nn.Linear(fusion_dim, fusion_dim // 2),
            nn.ReLU(inplace=True),
            nn.Dropout(p=0.3),
        )
        
        # Task heads
        self.malignancy_head = nn.Linear(fusion_dim // 2, num_classes)
        self.risk_head = nn.Sequential(
            nn.Linear(fusion_dim // 2, risk_years),
            nn.Sigmoid()  # per-year probability in [0,1]
        )
    
    def forward(self, volume, patch):

7.1 DICOM → Hounsfield Units Conversion
import numpy as np
import pydicom, SimpleITK as sitk
def load_dicom_volume(dicom_dir: str) -> np.ndarray:
    """Load DICOM series and return HU-calibrated volume."""
    reader = sitk.ImageSeriesReader()
    files  = reader.GetGDCMSeriesFileNames(dicom_dir)
    reader.SetFileNames(files)
    image  = reader.Execute()
    
    vol_hu = sitk.GetArrayFromImage(image).astype(np.float32)
    # HU = pixel_value * slope + intercept (already applied by SimpleITK)
    vol_hu = np.clip(vol_hu, -1000, 400)    # lung window
    vol_hu = (vol_hu + 1000) / 1400.0       # normalize → [0, 1]
    return vol_hu, image.GetSpacing()        # (volume, (x,y,z) mm)
def resample_to_isotropic(volume, orig_spacing, target_mm=1.0):
    """Resample volume to 1.0mm isotropic using linear interpolation."""
    scale = [s / target_mm for s in orig_spacing]
    new_shape = [int(round(d * sc))
                 for d, sc in zip(volume.shape, scale)]
    return zoom(volume, scale, order=1)  # scipy.ndimage.zoom
def extract_nodule_patch(volume, coord_zyx, patch_size=64):
    """Extract 64^3 patch centred on nodule coordinate."""
    z, y, x = coord_zyx
    h = patch_size // 2
    zs, ze = max(0,z-h), min(volume.shape[0], z+h)
    ys, ye = max(0,y-h), min(volume.shape[1], y+h)
    xs, xe = max(0,x-h), min(volume.shape[2], x+h)
    patch = volume[zs:ze, ys:ye, xs:xe]
    # Pad if near border
    pad = [(h-(z-zs), h-(ze-z)),
           (h-(y-ys), h-(ye-y)),
           (h-(x-xs), h-(xe-x))]
    return np.pad(patch, pad, mode='constant', constant_values=0)

import numpy as np
def build_lidc_dataset(cache_dir="./lidc_cache"):
    records = []
    for scan in pl.query(pl.Scan).all():
        vol = scan.to_volume(verbose=False)     # HU numpy array
        vol_norm = np.clip(vol,-1000,400)
        vol_norm = (vol_norm+1000)/1400.0
        
        anns = scan.cluster_annotations()
        for ann_group in anns:
            # Consensus centroid
            centroid = np.mean(
                [a.centroid for a in ann_group], axis=0).astype(int)
            
            # Consensus malignancy score (1–5)
            mal_score = np.mean(
                [a.malignancy for a in ann_group if a.malignancy])
            label = 1 if mal_score >= 3.0 else 0
            
            patch = extract_nodule_patch(vol_norm, centroid, 64)
            records.append({
                "scan_id": scan.patient_id,
                "patch"  : patch.astype(np.float32),
                "label"  : label,
                "score"  : mal_score
            })
    return records    # ~2,700 nodule patches

import torch, torch.nn as nn, torch.nn.functional as F
# Weighted Cross-Entropy for malignancy (handles class imbalance)
class WeightedCELoss(nn.Module):
    def __init__(self, pos_weight=3.0):  # 3:1 imbalance compensation
        super().__init__()
        w = torch.tensor([1.0, pos_weight])
        self.ce = nn.CrossEntropyLoss(weight=w)
    def forward(self, logits, labels):
        return self.ce(logits, labels)
# Binary Cross-Entropy for 6-year risk (each year is independent)
risk_loss_fn = nn.BCELoss()
# Combined loss
def total_loss(mal_logits, risk_preds,
               mal_labels, risk_labels,
               alpha=0.7, beta=0.3):
    L_mal  = WeightedCELoss()(mal_logits, mal_labels)
    L_risk = risk_loss_fn(risk_preds, risk_labels.float())
    return alpha * L_mal + beta * L_risk

Minimal Training Loop:
def train_epoch(model, loader, optimizer, device):
    model.train()
    total_loss = 0
    for batch in loader:
        volume = batch["volume"].to(device)   # (B,1,D,H,W)
        patch  = batch["patch"].to(device)    # (B,1,64,64,64)
        labels = batch["label"].to(device)    # (B,)
        risk_y = batch["risk_labels"].to(device) # (B,6) or zeros
        optimizer.zero_grad()
        mal_logits, risk_preds = model(volume, patch)
        loss = total_loss(mal_logits, risk_preds, labels, risk_y)
        loss.backward()
        torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)
        optimizer.step()
        total_loss += loss.item()
    return total_loss / len(loader)

    results = []
    model.eval()
    with torch.no_grad():
        for coord in nodule_coords_list:
            patch = extract_nodule_patch(vol_iso, coord, 64)
            p_t   = torch.from_numpy(patch).unsqueeze(0).unsqueeze(0).to(device)
            
            mal_logits, risk_scores = model(volume_tensor, p_t)
            mal_prob  = torch.softmax(mal_logits, dim=1)[0, 1].item()
            risk_list = risk_scores[0].cpu().tolist()
            
            results.append({
                "nodule_coord"     : coord,
                "malignancy_prob"  : round(mal_prob, 4),
                "risk_yr1_to_yr6"  : [round(r,4) for r in risk_list],
                "verdict"          : "MALIGNANT" if mal_prob > 0.5 else "BENIGN"
            })
    return results
